5/// The IBM extended-precision (double-double) floating-point type.
6// FIXME(ppcf128) improve the docs.
7#[doc(alias = "__ibm128")]
8#[doc(alias = "doubledouble")]
9#[doc(alias = "f64f64")]
10#[unstable(feature = "powerpc_ppcf128", issue = "163323")]
11#[allow(non_camel_case_types)]
12#[doc(cfg(any(target_arch = "powerpc", target_arch = "powerpc64")))]
13#[derive(Clone, Copy)]
14#[repr(transparent)]
15pub struct ppcf128(crate::num::imp::ppcf128::ppcf128);
16
17#[unstable(feature = "powerpc_ppcf128", issue = "163323")]
18impl crate::fmt::Debug for ppcf128 {
19    fn fmt(&self, f: &mut crate::fmt::Formatter<'_>) -> crate::fmt::Result {
20        let (hi, lo) = self.to_components();
21        write!(f, "ppcf128({hi:?}, {lo:?})")
22    }
23}
24
25impl ppcf128 {
26    /// Returns the memory representation of this floating point number as a byte array in
27    /// native byte order.
28    #[unstable(feature = "powerpc_ppcf128", issue = "163323")]
29    #[inline]
30    pub const fn to_ne_bytes(self) -> [u8; 16] {
31        // SAFETY: every bit pattern of a `ppcf128` is a valid `[u8; 16]`.
32        unsafe { crate::mem::transmute::<ppcf128, [u8; 16]>(self) }
33    }
34
35    /// Returns the large and small component of this floating point number.
36    #[unstable(feature = "powerpc_ppcf128", issue = "163323")]
37    #[rustc_const_unstable(feature = "powerpc_ppcf128", issue = "163323")]
38    #[inline]
39    pub const fn to_components(self) -> (f64, f64) {
40        // SAFETY: every bit pattern of a `ppcf128` is a valid `[f64; 2]`.
41        let [large, small] = unsafe { crate::mem::transmute::<ppcf128, [f64; 2]>(self) };
42        (large, small)
43    }
44}
