24/// Byte-wise lookup of indices into a vector in registers.
25///
26/// The xperm8 instruction operates on bytes. The rs1 register contains a vector of 4
27/// 8-bit elements. The rs2 register contains a vector of 4 8-bit indexes. The result is
28/// each element in rs2 replaced by the indexed element in rs1, or zero if the index into rs2
29/// is out of bounds.
30///
31/// Source: RISC-V Cryptography Extensions Volume I: Scalar & Entropy Source Instructions
32///
33/// Version: v1.0.1
34///
35/// Section: 3.47
36#[unstable(feature = "riscv_ext_intrinsics", issue = "114544")]
37#[target_feature(enable = "zbkx")]
38#[cfg_attr(test, assert_instr(xperm8))]
39#[inline]
40pub fn xperm8_32(rs1: u32, rs2: u32) -> u32 {
41    unsafe { _xperm8_32(rs1, rs2) }
42}
