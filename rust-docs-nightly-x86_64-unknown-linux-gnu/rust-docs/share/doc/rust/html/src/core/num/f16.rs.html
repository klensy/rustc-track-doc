311    /// Returns the floating point category of the number. If only one property
312    /// is going to be tested, it is generally faster to use the specific
313    /// predicate instead.
314    ///
315    /// ```
316    /// #![feature(f16)]
317    /// # #[cfg(target_has_reliable_f16)] {
318    ///
319    /// use std::num::FpCategory;
320    ///
321    /// let num = 12.4_f16;
322    /// let inf = f16::INFINITY;
323    ///
324    /// assert_eq!(num.classify(), FpCategory::Normal);
325    /// assert_eq!(inf.classify(), FpCategory::Infinite);
326    /// # }
327    /// ```
328    #[inline]
329    #[unstable(feature = "f16", issue = "116909")]
330    #[must_use]
331    pub const fn classify(self) -> FpCategory {
332        let b = self.to_bits();
333        match (b & Self::MANTISSA_MASK, b & Self::EXPONENT_MASK) {
334            (0, Self::EXPONENT_MASK) => FpCategory::Infinite,
335            (_, Self::EXPONENT_MASK) => FpCategory::Nan,
336            (0, 0) => FpCategory::Zero,
337            (_, 0) => FpCategory::Subnormal,
338            _ => FpCategory::Normal,
339        }
340    }

342    /// Returns `true` if `self` has a positive sign, including `+0.0`, NaNs with
343    /// positive sign bit and positive infinity.
344    ///
345    /// Note that IEEE 754 doesn't assign any meaning to the sign bit in case of
346    /// a NaN, and as Rust doesn't guarantee that the bit pattern of NaNs are

348    /// a NaN might produce an unexpected or non-portable result. See the [specification

370    /// Returns `true` if `self` has a negative sign, including `-0.0`, NaNs with
371    /// negative sign bit and negative infinity.
372    ///
373    /// Note that IEEE 754 doesn't assign any meaning to the sign bit in case of
374    /// a NaN, and as Rust doesn't guarantee that the bit pattern of NaNs are
375    /// conserved over arithmetic operations, the result of `is_sign_negative` on
376    /// a NaN might produce an unexpected or non-portable result. See the [specification
377    /// of NaN bit patterns](f32#nan-bit-patterns) for more info. Use `self.signum() == -1.0`
378    /// if you need fully portable behavior (evaluates to `false` for all NaNs).
379    ///
380    /// ```
381    /// #![feature(f16)]
382    /// # #[cfg(target_has_reliable_f16)] {
383    ///
384    /// let f = 7.0_f16;
385    /// let g = -7.0_f16;
386    ///
387    /// assert!(!f.is_sign_negative());
388    /// assert!(g.is_sign_negative());
389    /// # }
390    /// ```
391    #[inline]
392    #[must_use]
393    #[unstable(feature = "f16", issue = "116909")]
394    pub const fn is_sign_negative(self) -> bool {
395        // IEEE754 says: isSignMinus(x) is true if and only if x has negative sign. isSignMinus
396        // applies to zeros and NaNs as well.
397        // SAFETY: This is just transmuting to get the sign bit, it's fine.
398        (self.to_bits() & (1 << 15)) != 0
399    }

401    /// Returns the least number greater than `self`.
402    ///
403    /// Let `TINY` be the smallest representable positive `f16`. Then,
404    ///  - if `self.is_nan()`, this returns `self`;
405    ///  - if `self` is [`NEG_INFINITY`], this returns [`MIN`];
406    ///  - if `self` is `-TINY`, this returns -0.0;
407    ///  - if `self` is -0.0 or +0.0, this returns `TINY`;
408    ///  - if `self` is [`MAX`] or [`INFINITY`], this returns [`INFINITY`];
409    ///  - otherwise the unique least value greater than `self` is returned.
410    ///
411    /// The identity `x.next_up() == -(-x).next_down()` holds for all non-NaN `x`. When `x`
412    /// is finite `x == x.next_up().next_down()` also holds.
413    ///
414    /// ```rust
415    /// #![feature(f16)]
416    /// # #[cfg(target_has_reliable_f16)] {
417    ///
418    /// // f16::EPSILON is the difference between 1.0 and the next number up.
419    /// assert_eq!(1.0f16.next_up(), 1.0 + f16::EPSILON);
420    /// // But not for most numbers.
421    /// assert!(0.1f16.next_up() < 0.1 + f16::EPSILON);
422    /// assert_eq!(4356f16.next_up(), 4360.0);
423    /// # }
424    /// ```
425    ///
426    /// This operation corresponds to IEEE-754 `nextUp`.
427    ///
428    /// [`NEG_INFINITY`]: Self::NEG_INFINITY
429    /// [`INFINITY`]: Self::INFINITY
430    /// [`MIN`]: Self::MIN
431    /// [`MAX`]: Self::MAX
432    #[inline]
433    #[doc(alias = "nextUp")]
434    #[unstable(feature = "f16", issue = "116909")]
435    #[must_use = "method returns a new number and does not mutate the original value"]
436    pub const fn next_up(self) -> Self {
437        // Some targets violate Rust's assumption of IEEE semantics, e.g. by flushing
438        // denormals to zero. This is in general unsound and unsupported, but here
439        // we do our best to still produce the correct result on such targets.
440        let bits = self.to_bits();
441        if self.is_nan() || bits == Self::INFINITY.to_bits() {
442            return self;
443        }
444
445        let abs = bits & !Self::SIGN_MASK;
446        let next_bits = if abs == 0 {
447            Self::TINY_BITS
448        } else if bits == abs {
449            bits + 1
450        } else {
451            bits - 1
452        };
453        Self::from_bits(next_bits)
454    }

456    /// Returns the greatest number less than `self`.
457    ///
458    /// Let `TINY` be the smallest representable positive `f16`. Then,
459    ///  - if `self.is_nan()`, this returns `self`;
460    ///  - if `self` is [`INFINITY`], this returns [`MAX`];
461    ///  - if `self` is `TINY`, this returns 0.0;
462    ///  - if `self` is -0.0 or +0.0, this returns `-TINY`;
463    ///  - if `self` is [`MIN`] or [`NEG_INFINITY`], this returns [`NEG_INFINITY`];
464    ///  - otherwise the unique greatest value less than `self` is returned.
465    ///
466    /// The identity `x.next_down() == -(-x).next_up()` holds for all non-NaN `x`. When `x`
467    /// is finite `x == x.next_down().next_up()` also holds.
468    ///
469    /// ```rust
470    /// #![feature(f16)]
471    /// # #[cfg(target_has_reliable_f16)] {
472    ///
473    /// let x = 1.0f16;
474    /// // Clamp value into range [0, 1).
475    /// let clamped = x.clamp(0.0, 1.0f16.next_down());
476    /// assert!(clamped < 1.0);
477    /// assert_eq!(clamped.next_up(), 1.0);
478    /// # }
479    /// ```
480    ///
481    /// This operation corresponds to IEEE-754 `nextDown`.
482    ///
483    /// [`NEG_INFINITY`]: Self::NEG_INFINITY
484    /// [`INFINITY`]: Self::INFINITY
485    /// [`MIN`]: Self::MIN
486    /// [`MAX`]: Self::MAX
487    #[inline]
488    #[doc(alias = "nextDown")]
489    #[unstable(feature = "f16", issue = "116909")]
490    #[must_use = "method returns a new number and does not mutate the original value"]
491    pub const fn next_down(self) -> Self {
492        // Some targets violate Rust's assumption of IEEE semantics, e.g. by flushing
493        // denormals to zero. This is in general unsound and unsupported, but here
494        // we do our best to still produce the correct result on such targets.
495        let bits = self.to_bits();
496        if self.is_nan() || bits == Self::NEG_INFINITY.to_bits() {
497            return self;
498        }
499
500        let abs = bits & !Self::SIGN_MASK;
501        let next_bits = if abs == 0 {
502            Self::NEG_TINY_BITS
503        } else if bits == abs {
504            bits - 1
505        } else {
506            bits + 1
507        };
508        Self::from_bits(next_bits)
509    }

715    /// Calculates the midpoint (average) between `self` and `rhs`.
716    ///
717    /// This returns NaN when *either* argument is NaN or if a combination of
718    /// +inf and -inf is provided as arguments.
719    ///
720    /// # Examples
721    ///
722    /// ```
723    /// #![feature(f16)]
724    /// # #[cfg(target_has_reliable_f16)] {
725    ///
726    /// assert_eq!(1f16.midpoint(4.0), 2.5);
727    /// assert_eq!((-5.5f16).midpoint(8.0), 1.25);
728    /// # }
729    /// ```
730    #[inline]
731    #[doc(alias = "average")]
732    #[unstable(feature = "f16", issue = "116909")]
733    #[rustc_const_unstable(feature = "f16", issue = "116909")]
734    #[must_use = "this returns the result of the operation, \
735                  without modifying the original"]
736    pub const fn midpoint(self, other: f16) -> f16 {
737        const HI: f16 = f16::MAX * 0.5;
738
739        let (a, b) = (self, other);
740        let abs_a = a.abs();
741        let abs_b = b.abs();
742
743        if abs_a <= HI && abs_b <= HI {
744            // Overflow is impossible
745            (a + b) * 0.5
746        } else {
747            (a * 0.5) + (b * 0.5)
748        }
749    }

908    /// Raw transmutation from `u16`.
909    ///
910    /// This is currently identical to `transmute::<u16, f16>(v)` on all platforms.
911    /// It turns out this is incredibly portable, for two reasons:
912    ///
913    /// * Floats and Ints have the same endianness on all supported platforms.
914    /// * IEEE 754 very precisely specifies the bit layout of floats.
915    ///
916    /// However there is one caveat: prior to the 2008 version of IEEE 754, how
917    /// to interpret the NaN signaling bit wasn't actually specified. Most platforms
918    /// (notably x86 and ARM) picked the interpretation that was ultimately
919    /// standardized in 2008, but some didn't (notably MIPS). As a result, all
920    /// signaling NaNs on MIPS are quiet NaNs on x86, and vice-versa.
921    ///
922    /// Rather than trying to preserve signaling-ness cross-platform, this
923    /// implementation favors preserving the exact bits. This means that
924    /// any payloads encoded in NaNs will be preserved even if the result of
925    /// this method is sent over the network from an x86 machine to a MIPS one.
926    ///
927    /// If the results of this method are only manipulated by the same
928    /// architecture that produced them, then there is no portability concern.
929    ///
930    /// If the input isn't NaN, then there is no portability concern.
931    ///
932    /// If you don't care about signalingness (very likely), then there is no
933    /// portability concern.
934    ///
935    /// Note that this function is distinct from `as` casting, which attempts to
936    /// preserve the *numeric* value, and not the bitwise value.
937    ///
938    /// ```
939    /// #![feature(f16)]
940    /// # #[cfg(target_has_reliable_f16)] {
941    ///
942    /// let v = f16::from_bits(0x4a40);
943    /// assert_eq!(v, 12.5);
944    /// # }
945    /// ```

952        // SAFETY: `u16` is a plain old datatype so we can always transmute from it.

954    }

1115    /// Returns the ordering between `self` and `other`.
1116    ///
1117    /// Unlike the standard partial comparison between floating point numbers,
1118    /// this comparison always produces an ordering in accordance to
1119    /// the `totalOrder` predicate as defined in the IEEE 754 (2008 revision)
1120    /// floating point standard. The values are ordered in the following sequence:
1121    ///
1122    /// - negative quiet NaN
1123    /// - negative signaling NaN
1124    /// - negative infinity
1125    /// - negative numbers
1126    /// - negative subnormal numbers
1127    /// - negative zero
1128    /// - positive zero
1129    /// - positive subnormal numbers
1130    /// - positive numbers
1131    /// - positive infinity
1132    /// - positive signaling NaN
1133    /// - positive quiet NaN.
1134    ///
1135    /// The ordering established by this function does not always agree with the
1136    /// [`PartialOrd`] and [`PartialEq`] implementations of `f16`. For example,
1137    /// they consider negative and positive zero equal, while `total_cmp`
1138    /// doesn't.
1139    ///
1140    /// The interpretation of the signaling NaN bit follows the definition in
1141    /// the IEEE 754 standard, which may not match the interpretation by some of
1142    /// the older, non-conformant (e.g. MIPS) hardware implementations.
1143    ///
1144    /// # Example
1145    ///
1146    /// ```
1147    /// #![feature(f16)]
1148    /// # #[cfg(target_has_reliable_f16)] {
1149    ///
1150    /// struct GoodBoy {
1151    ///     name: &'static str,
1152    ///     weight: f16,

1160    ///     GoodBoy { name: "Abs. Unit", weight: f16::NAN },
1161    ///     GoodBoy { name: "Floaty", weight: -5.0 },
1162    /// ];
1163    ///
1164    /// bois.sort_by(|a, b| a.weight.total_cmp(&b.weight));
1165    ///
1166    /// // `f16::NAN` could be positive or negative, which will affect the sort order.
1167    /// if f16::NAN.is_sign_negative() {
1168    ///     bois.into_iter().map(|b| b.weight)
1169    ///         .zip([f16::NAN, -5.0, 0.1, 10.0, 99.0, f16::INFINITY].iter())
1170    ///         .for_each(|(a, b)| assert_eq!(a.to_bits(), b.to_bits()))
1171    /// } else {
1172    ///     bois.into_iter().map(|b| b.weight)
1173    ///         .zip([-5.0, 0.1, 10.0, 99.0, f16::INFINITY, f16::NAN].iter())
1174    ///         .for_each(|(a, b)| assert_eq!(a.to_bits(), b.to_bits()))
1175    /// }
1176    /// # }

1179    #[must_use]
1180    #[unstable(feature = "f16", issue = "116909")]
1181    #[rustc_const_unstable(feature = "const_cmp", issue = "143800")]
1182    pub const fn total_cmp(&self, other: &Self) -> crate::cmp::Ordering {
1183        let mut left = self.to_bits() as i16;
1184        let mut right = other.to_bits() as i16;
1185
1186        // In case of negatives, flip all the bits except the sign
1187        // to achieve a similar layout as two's complement integers
1188        //
1189        // Why does this work? IEEE 754 floats consist of three fields:
1190        // Sign bit, exponent and mantissa. The set of exponent and mantissa
1191        // fields as a whole have the property that their bitwise order is
1192        // equal to the numeric magnitude where the magnitude is defined.
1193        // The magnitude is not normally defined on NaN values, but
1194        // IEEE 754 totalOrder defines the NaN values also to follow the
1195        // bitwise order. This leads to order explained in the doc comment.
1196        // However, the representation of magnitude is the same for negative
1197        // and positive numbers – only the sign bit is different.
1198        // To easily compare the floats as signed integers, we need to
1199        // flip the exponent and mantissa bits in case of negative numbers.
1200        // We effectively convert the numbers to "two's complement" form.
1201        //
1202        // To do the flipping, we construct a mask and XOR against it.
1203        // We branchlessly calculate an "all-ones except for the sign bit"
1204        // mask from negative-signed values: right shifting sign-extends
1205        // the integer, so we "fill" the mask with sign bits, and then
1206        // convert to unsigned to push one more zero bit.
1207        // On positive values, the mask is all zeros, so it's a no-op.
1208        left ^= (((left >> 15) as u16) >> 1) as i16;
1209        right ^= (((right >> 15) as u16) >> 1) as i16;
1210
1211        left.cmp(&right)
1212    }

1782    /// Calculates Euclidean division, the matching method for `rem_euclid`.
1783    ///
1784    /// This computes the integer `n` such that
1785    /// `self = n * rhs + self.rem_euclid(rhs)`.
1786    /// In other words, the result is `self / rhs` rounded to the integer `n`
1787    /// such that `self >= n * rhs`.
1788    ///
1789    /// # Precision
1790    ///
1791    /// The result of this operation is guaranteed to be the rounded
1792    /// infinite-precision result.
1793    ///
1794    /// # Examples
1795    ///
1796    /// ```
1797    /// #![feature(f16)]
1798    /// # #[cfg(target_has_reliable_f16)] {
1799    ///
1800    /// let a: f16 = 7.0;
1801    /// let b = 4.0;
1802    /// assert_eq!(a.div_euclid(b), 1.0); // 7.0 > 4.0 * 1.0
1803    /// assert_eq!((-a).div_euclid(b), -2.0); // -7.0 >= 4.0 * -2.0
1804    /// assert_eq!(a.div_euclid(-b), -1.0); // 7.0 >= -4.0 * -1.0
1805    /// assert_eq!((-a).div_euclid(-b), 2.0); // -7.0 >= -4.0 * 2.0
1806    /// # }
1807    /// ```
1808    #[inline]
1809    #[rustc_allow_incoherent_impl]
1810    #[unstable(feature = "f16", issue = "116909")]
1811    #[must_use = "method returns a new number and does not mutate the original value"]
1812    pub fn div_euclid(self, rhs: f16) -> f16 {
1813        let q = (self / rhs).trunc();
1814        if self % rhs < 0.0 {
1815            return if rhs > 0.0 { q - 1.0 } else { q + 1.0 };
1816        }
1817        q
1818    }

1820    /// Calculates the least nonnegative remainder of `self` when
1821    /// divided by `rhs`.
1822    ///
1823    /// In particular, the return value `r` satisfies `0.0 <= r < rhs.abs()` in
1824    /// most cases. However, due to a floating point round-off error it can
1825    /// result in `r == rhs.abs()`, violating the mathematical definition, if
1826    /// `self` is much smaller than `rhs.abs()` in magnitude and `self < 0.0`.
1827    /// This result is not an element of the function's codomain, but it is the
1828    /// closest floating point number in the real numbers and thus fulfills the
1829    /// property `self == self.div_euclid(rhs) * rhs + self.rem_euclid(rhs)`
1830    /// approximately.
1831    ///
1832    /// # Precision
1833    ///
1834    /// The result of this operation is guaranteed to be the rounded
1835    /// infinite-precision result.
1836    ///
1837    /// # Examples
1838    ///
1839    /// ```
1840    /// #![feature(f16)]
1841    /// # #[cfg(target_has_reliable_f16)] {
1842    ///
1843    /// let a: f16 = 7.0;
1844    /// let b = 4.0;
1845    /// assert_eq!(a.rem_euclid(b), 3.0);
1846    /// assert_eq!((-a).rem_euclid(b), 1.0);
1847    /// assert_eq!(a.rem_euclid(-b), 3.0);
1848    /// assert_eq!((-a).rem_euclid(-b), 1.0);
1849    /// // limitation due to round-off error
1850    /// assert!((-f16::EPSILON).rem_euclid(3.0) != 0.0);
1851    /// # }
1852    /// ```
1853    #[inline]
1854    #[rustc_allow_incoherent_impl]
1855    #[doc(alias = "modulo", alias = "mod")]
1856    #[unstable(feature = "f16", issue = "116909")]
1857    #[must_use = "method returns a new number and does not mutate the original value"]
1858    pub fn rem_euclid(self, rhs: f16) -> f16 {
1859        let r = self % rhs;
1860        if r < 0.0 { r + rhs.abs() } else { r }
1861    }
