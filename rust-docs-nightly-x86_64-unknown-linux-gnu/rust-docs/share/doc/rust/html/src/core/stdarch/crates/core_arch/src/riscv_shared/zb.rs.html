12/// Bitwise OR-Combine, byte granule
13///
14/// Combines the bits within every byte through a reciprocal bitwise logical OR. This sets the bits of each byte in
15/// the result rd to all zeros if no bit within the respective byte of rs is set, or to all ones if any bit within the
16/// respective byte of rs is set.
17///
18/// Source: RISC-V Bit-Manipulation ISA-extensions
19///
20/// Version: v1.0.0
21///
22/// Section: 2.24
23#[unstable(feature = "riscv_ext_intrinsics", issue = "114544")]
24#[target_feature(enable = "zbb")]
25#[cfg_attr(test, assert_instr(orc.b))]
26#[inline]
27pub fn orc_b_32(rs: u32) -> u32 {
28    unsafe { _orc_b_32(rs) }
29}
