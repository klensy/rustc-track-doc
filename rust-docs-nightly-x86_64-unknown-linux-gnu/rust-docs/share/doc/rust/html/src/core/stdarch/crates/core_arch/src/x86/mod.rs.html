8/// The x87 80-bit extended-precision floating-point type.
9///
10/// FIXME: extend the documentation.
11#[doc(alias = "long double")]
12#[doc(alias = "x86_fp80")]
13#[doc(alias = "__float80")]

16#[doc(cfg(any(target_arch = "x86", target_arch = "x86_64")))]
17#[derive(Clone, Copy)]

19pub struct x87_f80(crate::num::imp::x87_f80::x87_f80);

47#[unstable(feature = "x87_f80", issue = "163982")]
48impl crate::fmt::Debug for x87_f80 {
49    #[inline]
50    fn fmt(&self, f: &mut crate::fmt::Formatter<'_>) -> crate::fmt::Result {
51        let mut bytes = [0u8; 16];
52        bytes[..10].copy_from_slice(&self.to_ne_bytes());
53        write!(f, "x87_f80({:#022x})", u128::from_ne_bytes(bytes))
54    }
55}
56
57impl x87_f80 {
58    /// Returns the memory representation of this floating point number as a byte array in
59    /// native byte order.
60    #[unstable(feature = "x87_f80", issue = "163982")]
61    #[inline]
62    pub const fn to_ne_bytes(self) -> [u8; 10] {
63        // SAFETY: the first 10 bytes are initialized.
64        unsafe { (&raw const self).cast::<[u8; 10]>().read() }
65    }
66
67    /// Creates a floating point value from its representation as a byte array in native endian.
68    #[unstable(feature = "x87_f80", issue = "163982")]
69    #[inline]
70    pub const fn from_ne_bytes(bytes: [u8; 10]) -> Self {
71        // SAFETY: we initialize the first 10 bytes, the remaining bytes remain uninitialized.
72        // That is exactly the layout of x87_f80.
73        unsafe {
74            let mut this = crate::mem::MaybeUninit::<x87_f80>::uninit();
75            this.as_mut_ptr().cast::<[u8; 10]>().write(bytes);
76            this.assume_init()
77        }
78    }
79}

84    /// 128-bit wide integer vector type, x86-specific
85    ///
86    /// This type is the same as the `__m128i` type defined by Intel,
87    /// representing a 128-bit SIMD register. Usage of this type typically
88    /// corresponds to the `sse` and up target features for x86/x86_64.
89    ///
90    /// Internally this type may be viewed as:
91    ///
92    /// * `i8x16` - sixteen `i8` variables packed together
93    /// * `i16x8` - eight `i16` variables packed together
94    /// * `i32x4` - four `i32` variables packed together
95    /// * `i64x2` - two `i64` variables packed together
96    ///
97    /// (as well as unsigned versions). Each intrinsic may interpret the
98    /// internal bits differently, check the documentation of the intrinsic
99    /// to see how it's being used.
100    ///
101    /// The in-memory representation of this type is the same as the one of an
102    /// equivalent array (i.e. the in-memory order of elements is the same, and
103    /// there is no padding); however, the alignment is different and equal to
104    /// the size of the type. Note that the ABI for function calls may *not* be
105    /// the same.
106    ///
107    /// Note that this means that an instance of `__m128i` typically just means
108    /// a "bag of bits" which is left up to interpretation at the point of use.
109    ///
110    /// Most intrinsics using `__m128i` are prefixed with `_mm_` and the
111    /// integer types tend to correspond to suffixes like "epi8" or "epi32".
112    ///
113    /// # Examples
114    ///
115    /// ```
116    /// # #[cfg(any(target_arch = "x86", target_arch = "x86_64"))] {
117    /// #[cfg(target_arch = "x86")]
118    /// use std::arch::x86::*;
119    /// #[cfg(target_arch = "x86_64")]
120    /// use std::arch::x86_64::*;
121    ///
122    /// # #[target_feature(enable = "sse2")]
123    /// # #[allow(unused_unsafe)] // temporary, to unstick CI
124    /// # unsafe fn foo() { unsafe {
125    /// let all_bytes_zero = _mm_setzero_si128();
126    /// let all_bytes_one = _mm_set1_epi8(1);
127    /// let four_i32 = _mm_set_epi32(1, 2, 3, 4);
128    /// # }}
129    /// # if is_x86_feature_detected!("sse2") { unsafe { foo() } }
130    /// # }
131    /// ```
132    #[doc(cfg(any(target_arch = "x86", target_arch = "x86_64")))]
133    pub struct __m128i(2 x i64);
134

598macro_rules! as_transmute {
599    ($from:ty => $as_from:ident, $($as_to:ident -> $to:ident),* $(,)?) => {
600        impl $from {$(
601            #[inline]
602            pub(crate) const fn $as_to(self) -> crate::core_arch::simd::$to {
603                unsafe { transmute(self) }
604            }
605        )*}
606        $(
607            impl crate::core_arch::simd::$to {
608                #[inline]
609                pub(crate) const fn $as_from(self) -> $from {
610                    unsafe { transmute(self) }
611                }
612            }
613        )*
614    };
615}
