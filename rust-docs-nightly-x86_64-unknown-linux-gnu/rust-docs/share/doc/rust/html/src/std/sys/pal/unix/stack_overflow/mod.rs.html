4cfg_select! {
5    // This is intentionally not enabled on iOS/tvOS/watchOS/visionOS, as it uses
6    // several symbols that might lead to rejections from the App Store, namely
7    // `sigaction`, `sigaltstack`, `sysctlbyname`, `mmap`, `munmap` and `mprotect`.
8    //
9    // This might be overly cautious, though it is also what Swift does (and they
10    // usually have fewer qualms about forwards compatibility, since the runtime
11    // is shipped with the OS):
12    // <https://github.com/apple/swift/blob/swift-5.10-RELEASE/stdlib/public/runtime/CrashHandlerMacOS.cpp>
13    //
14    // miri doesn't model signals nor stack overflows and this code has some
15    // synchronization properties that we don't want to expose to user code,
16    // hence we disable it on miri.
17    all(
18        not(miri),
19        any(
20            target_os = "linux",
21            target_os = "freebsd",
22            target_os = "hurd",
23            target_os = "macos",
24            target_os = "netbsd",
25            target_os = "openbsd",
26            target_os = "solaris",
27            target_os = "illumos",
28        )
29    ) => {
30        mod handler_signal;
31        mod thread_info;
32        use handler_signal as imp;
33    }
34    target_os = "cygwin" => {
35        mod handler_cygwin;
36        use handler_cygwin as imp;
37    }
38    _ => {
39        mod handler_none;
40        use handler_none as imp;
41    }
42}
