12use crate::sys::net::netc as c;
13use crate::sys_common::{FromInner, IntoInner};

15impl IntoInner<c::in_addr> for Ipv4Addr {
16    #[inline]
17    fn into_inner(self) -> c::in_addr {
18        // `s_addr` is stored as BE on all machines and the array is in BE order.
19        // So the native endian conversion method is used so that it's never swapped.
20        c::in_addr { s_addr: u32::from_ne_bytes(self.octets()) }
21    }
22}
23impl FromInner<c::in_addr> for Ipv4Addr {
24    fn from_inner(addr: c::in_addr) -> Ipv4Addr {
25        Ipv4Addr::from(addr.s_addr.to_ne_bytes())
26    }
27}
28
29impl IntoInner<c::in6_addr> for Ipv6Addr {
30    fn into_inner(self) -> c::in6_addr {
31        c::in6_addr { s6_addr: self.octets() }
32    }
33}
34impl FromInner<c::in6_addr> for Ipv6Addr {
35    #[inline]
36    fn from_inner(addr: c::in6_addr) -> Ipv6Addr {
37        Ipv6Addr::from(addr.s6_addr)
38    }
