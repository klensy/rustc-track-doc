1//! Traits, helpers, and type definitions for core I/O functionality.
2//!
3//! The `std::io` module contains a number of common things you'll need
4//! when doing input and output. The most core part of this module is
5//! the [`Read`] and [`Write`] traits, which provide the
6//! most general interface for reading and writing input and output.
7//!
8//! ## Read and Write
9//!
10//! Because they are traits, [`Read`] and [`Write`] are implemented by a number
11//! of other types, and you can implement them for your types too. As such,
12//! you'll see a few different types of I/O throughout the documentation in
13//! this module: [`File`]s, [`TcpStream`]s, and sometimes even [`Vec<T>`]s. For
14//! example, [`Read`] adds a [`read`][`Read::read`] method, which we can use on
15//! [`File`]s:
16//!
17//! ```no_run
18//! use std::io;
19//! use std::io::prelude::*;
20//! use std::fs::File;
21//!
22//! fn main() -> io::Result<()> {
23//!     let mut f = File::open("foo.txt")?;
24//!     let mut buffer = [0; 10];
25//!
26//!     // read up to 10 bytes
27//!     let n = f.read(&mut buffer)?;
28//!
29//!     println!("The bytes: {:?}", &buffer[..n]);
30//!     Ok(())
31//! }
32//! ```
33//!
34//! [`Read`] and [`Write`] are so important, implementors of the two traits have a
35//! nickname: readers and writers. So you'll sometimes see 'a reader' instead
36//! of 'a type that implements the [`Read`] trait'. Much easier!
37//!
38//! ## Seek and BufRead
39//!
40//! Beyond that, there are two important traits that are provided: [`Seek`]
41//! and [`BufRead`]. Both of these build on top of a reader to control
42//! how the reading happens. [`Seek`] lets you control where the next byte is
43//! coming from:
44//!
45//! ```no_run
46//! use std::io;
47//! use std::io::prelude::*;
48//! use std::io::SeekFrom;
49//! use std::fs::File;
50//!
51//! fn main() -> io::Result<()> {
52//!     let mut f = File::open("foo.txt")?;
53//!     let mut buffer = [0; 10];
54//!
55//!     // skip to the last 10 bytes of the file
56//!     f.seek(SeekFrom::End(-10))?;
57//!
58//!     // read up to 10 bytes
59//!     let n = f.read(&mut buffer)?;
60//!
61//!     println!("The bytes: {:?}", &buffer[..n]);
62//!     Ok(())
63//! }
64//! ```
65//!
66//! [`BufRead`] uses an internal buffer to provide a number of other ways to read, but
67//! to show it off, we'll need to talk about buffers in general. Keep reading!
68//!
69//! ## BufReader and BufWriter
70//!
71//! Byte-based interfaces are unwieldy and can be inefficient, as we'd need to be
72//! making near-constant calls to the operating system. To help with this,
73//! `std::io` comes with two structs, [`BufReader`] and [`BufWriter`], which wrap
74//! readers and writers. The wrapper uses a buffer, reducing the number of
75//! calls and providing nicer methods for accessing exactly what you want.
76//!
77//! For example, [`BufReader`] works with the [`BufRead`] trait to add extra
78//! methods to any reader:
79//!
80//! ```no_run
81//! use std::io;
82//! use std::io::prelude::*;
83//! use std::io::BufReader;
84//! use std::fs::File;
85//!
86//! fn main() -> io::Result<()> {
87//!     let f = File::open("foo.txt")?;
88//!     let mut reader = BufReader::new(f);
89//!     let mut buffer = String::new();
90//!
91//!     // read a line into buffer
92//!     reader.read_line(&mut buffer)?;
93//!
94//!     println!("{buffer}");
95//!     Ok(())
96//! }
97//! ```
98//!
99//! [`BufWriter`] doesn't add any new ways of writing; it just buffers every call
100//! to [`write`][`Write::write`]:
101//!
102//! ```no_run
103//! use std::io;
104//! use std::io::prelude::*;
105//! use std::io::BufWriter;
106//! use std::fs::File;
107//!
108//! fn main() -> io::Result<()> {
109//!     let f = File::create("foo.txt")?;
110//!     {
111//!         let mut writer = BufWriter::new(f);
112//!
113//!         // write a byte to the buffer
114//!         writer.write(&[42])?;
115//!
116//!     } // the buffer is flushed once writer goes out of scope
117//!
118//!     Ok(())
119//! }
120//! ```
121//!
122//! ## Standard input and output
123//!
124//! A very common source of input is standard input:
125//!
126//! ```no_run
127//! use std::io;
128//!
129//! fn main() -> io::Result<()> {
130//!     let mut input = String::new();
131//!
132//!     io::stdin().read_line(&mut input)?;
133//!
134//!     println!("You typed: {}", input.trim());
135//!     Ok(())
136//! }
137//! ```
138//!
139//! Note that you cannot use the [`?` operator] in functions that do not return
140//! a [`Result<T, E>`][`Result`]. Instead, you can call [`.unwrap()`]
141//! or `match` on the return value to catch any possible errors:
142//!
143//! ```no_run
144//! use std::io;
145//!
146//! let mut input = String::new();
147//!
148//! io::stdin().read_line(&mut input).unwrap();
149//! ```
150//!
151//! And a very common source of output is standard output:
152//!
153//! ```no_run
154//! use std::io;
155//! use std::io::prelude::*;
156//!
157//! fn main() -> io::Result<()> {
158//!     io::stdout().write(&[42])?;
159//!     Ok(())
160//! }
161//! ```
162//!
163//! Of course, using [`io::stdout`] directly is less common than something like
164//! [`println!`].
165//!
166//! ## Iterator types
167//!
168//! A large number of the structures provided by `std::io` are for various
169//! ways of iterating over I/O. For example, [`Lines`] is used to split over
170//! lines:
171//!
172//! ```no_run
173//! use std::io;
174//! use std::io::prelude::*;
175//! use std::io::BufReader;
176//! use std::fs::File;
177//!
178//! fn main() -> io::Result<()> {
179//!     let f = File::open("foo.txt")?;
180//!     let reader = BufReader::new(f);
181//!
182//!     for line in reader.lines() {
183//!         println!("{}", line?);
184//!     }
185//!     Ok(())
186//! }
187//! ```
188//!
189//! ## Functions
190//!
191//! There are a number of [functions][functions-list] that offer access to various
192//! features. For example, we can use three of these functions to copy everything
193//! from standard input to standard output:
194//!
195//! ```no_run
196//! use std::io;
197//!
198//! fn main() -> io::Result<()> {
199//!     io::copy(&mut io::stdin(), &mut io::stdout())?;
200//!     Ok(())
201//! }
202//! ```
203//!
204//! [functions-list]: #functions-1
205//!
206//! ## io::Result
207//!
208//! Last, but certainly not least, is [`io::Result`]. This type is used
209//! as the return type of many `std::io` functions that can cause an error, and
210//! can be returned from your own functions as well. Many of the examples in this
211//! module use the [`?` operator]:
212//!
213//! ```
214//! use std::io;
215//!
216//! fn read_input() -> io::Result<()> {
217//!     let mut input = String::new();
218//!
219//!     io::stdin().read_line(&mut input)?;
220//!
221//!     println!("You typed: {}", input.trim());
222//!
223//!     Ok(())
224//! }
225//! ```
226//!
227//! The return type of `read_input()`, [`io::Result<()>`][`io::Result`], is a very
228//! common type for functions which don't have a 'real' return value, but do want to
229//! return errors if they happen. In this case, the only purpose of this function is
230//! to read the line and print it, so we use `()`.
231//!
232//! ## Platform-specific behavior
233//!
234//! Many I/O functions throughout the standard library are documented to indicate
235//! what various library or syscalls they are delegated to. This is done to help
236//! applications both understand what's happening under the hood as well as investigate
237//! any possibly unclear semantics. Note, however, that this is informative, not a binding
238//! contract. The implementation of many of these functions are subject to change over
239//! time and may call fewer or more syscalls/library functions.
240//!
241//! ## I/O Safety
242//!
243//! Rust follows an I/O safety discipline that is comparable to its memory safety discipline. This
244//! means that file descriptors can be *exclusively owned*. (Here, "file descriptor" is meant to
245//! subsume similar concepts that exist across a wide range of operating systems even if they might
246//! use a different name, such as "handle".) An exclusively owned file descriptor is one that no
247//! other code is allowed to access in any way, but the owner is allowed to access and even close
248//! it any time. A type that owns its file descriptor should usually close it in its `drop`
249//! function. Types like [`File`] own their file descriptor. Similarly, file descriptors
250//! can be *borrowed*, granting the temporary right to perform operations on this file descriptor.
251//! This indicates that the file descriptor will not be closed for the lifetime of the borrow, but
252//! it does *not* imply any right to close this file descriptor, since it will likely be owned by
253//! someone else.
254//!
255//! The platform-specific parts of the Rust standard library expose types that reflect these
256//! concepts, see [`os::unix`] and [`os::windows`].
257//!
258//! To uphold I/O safety, it is crucial that no code acts on file descriptors it does not own or
259//! borrow, and no code closes file descriptors it does not own. In other words, a safe function
260//! that takes a regular integer, treats it as a file descriptor, and acts on it, is *unsound*.
261//!
262//! Not upholding I/O safety and acting on a file descriptor without proof of ownership can lead to
263//! misbehavior and even Undefined Behavior in code that relies on ownership of its file
264//! descriptors: a closed file descriptor could be re-allocated, so the original owner of that file
265//! descriptor is now working on the wrong file. Some code might even rely on fully encapsulating
266//! its file descriptors with no operations being performed by any other part of the program.
267//!
268//! Note that exclusive ownership of a file descriptor does *not* imply exclusive ownership of the
269//! underlying kernel object that the file descriptor references (also called "open file description" on
270//! some operating systems). File descriptors basically work like [`Arc`]: when you receive an owned
271//! file descriptor, you cannot know whether there are any other file descriptors that reference the
272//! same kernel object. However, when you create a new kernel object, you know that you are holding
273//! the only reference to it. Just be careful not to lend it to anyone, since they can obtain a
274//! clone and then you can no longer know what the reference count is! In that sense, [`OwnedFd`] is
275//! like `Arc` and [`BorrowedFd<'a>`] is like `&'a Arc` (and similar for the Windows types). In
276//! particular, given a `BorrowedFd<'a>`, you are not allowed to close the file descriptor -- just
277//! like how, given a `&'a Arc`, you are not allowed to decrement the reference count and
278//! potentially free the underlying object. There is no equivalent to `Box` for file descriptors in
279//! the standard library (that would be a type that guarantees that the reference count is `1`),
280//! however, it would be possible for a crate to define a type with those semantics.
281//!

364fn read_until<R: BufRead + ?Sized>(r: &mut R, delim: u8, buf: &mut Vec<u8>) -> Result<usize> {
365    let mut read = 0;
366    loop {
367        let (done, used) = {
368            let available = match r.fill_buf() {
369                Ok(n) => n,
370                Err(ref e) if e.is_interrupted() => continue,
371                Err(e) => return Err(e),
372            };
373            match memchr::memchr(delim, available) {
374                Some(i) => {
375                    buf.extend_from_slice(&available[..=i]);
376                    (true, i + 1)
377                }
378                None => {
379                    buf.extend_from_slice(available);
380                    (false, available.len())
381                }
382            }
383        };
384        r.consume(used);
385        read += used;
386        if done || used == 0 {
387            return Ok(read);
388        }
389    }
390}

392fn skip_until<R: BufRead + ?Sized>(r: &mut R, delim: u8) -> Result<usize> {
393    let mut read = 0;
394    loop {
395        let (done, used) = {
396            let available = match r.fill_buf() {
397                Ok(n) => n,
398                Err(ref e) if e.kind() == ErrorKind::Interrupted => continue,
399                Err(e) => return Err(e),
400            };
401            match memchr::memchr(delim, available) {
402                Some(i) => (true, i + 1),
403                None => (false, available.len()),
404            }
405        };
406        r.consume(used);
407        read += used;
408        if done || used == 0 {
409            return Ok(read);
410        }
411    }
412}

414/// A `BufRead` is a type of `Read`er which has an internal buffer, allowing it
415/// to perform extra ways of reading.
416///
417/// For example, reading line-by-line is inefficient without using a buffer, so
418/// if you want to read by line, you'll need `BufRead`, which includes a
419/// [`read_line`] method as well as a [`lines`] iterator.
420///
421/// # Examples
422///
423/// A locked standard input implements `BufRead`:
424///
425/// ```no_run
426/// use std::io;
427/// use std::io::prelude::*;
428///
429/// let stdin = io::stdin();
430/// for line in stdin.lock().lines() {
431///     println!("{}", line?);
432/// }
433/// # std::io::Result::Ok(())
434/// ```
435///
436/// If you have something that implements [`Read`], you can use the [`BufReader`
437/// type][`BufReader`] to turn it into a `BufRead`.
438///
439/// For example, [`File`] implements [`Read`], but not `BufRead`.
440/// [`BufReader`] to the rescue!
441///
442/// [`File`]: crate::fs::File
443/// [`read_line`]: BufRead::read_line
444/// [`lines`]: BufRead::lines
445///
446/// ```no_run
447/// use std::io::{self, BufReader};
448/// use std::io::prelude::*;
449/// use std::fs::File;
450///
451/// fn main() -> io::Result<()> {
452///     let f = File::open("foo.txt")?;
453///     let f = BufReader::new(f);
454///
455///     for line in f.lines() {
456///         let line = line?;
457///         println!("{line}");
458///     }
459///
460///     Ok(())
461/// }
462/// ```
463#[stable(feature = "rust1", since = "1.0.0")]
464#[cfg_attr(not(test), rustc_diagnostic_item = "IoBufRead")]
465pub trait BufRead: Read {
466    /// Returns the contents of the internal buffer, filling it with more data, via `Read` methods, if empty.
467    ///
468    /// This is a lower-level method and is meant to be used together with [`consume`],
469    /// which can be used to mark bytes that should not be returned by subsequent calls to `read`.
470    ///
471    /// [`consume`]: BufRead::consume
472    ///
473    /// Returns an empty buffer when the stream has reached EOF.
474    ///
475    /// # Errors
476    ///
477    /// This function will return an I/O error if a `Read` method was called, but returned an error.
478    ///
479    /// # Examples
480    ///
481    /// A locked standard input implements `BufRead`:
482    ///
483    /// ```no_run
484    /// use std::io;
485    /// use std::io::prelude::*;
486    ///
487    /// let stdin = io::stdin();
488    /// let mut stdin = stdin.lock();
489    ///
490    /// let buffer = stdin.fill_buf()?;
491    ///
492    /// // work with buffer
493    /// println!("{buffer:?}");
494    ///
495    /// // mark the bytes we worked with as read
496    /// let length = buffer.len();
497    /// stdin.consume(length);
498    /// # std::io::Result::Ok(())
499    /// ```
500    #[stable(feature = "rust1", since = "1.0.0")]
501    fn fill_buf(&mut self) -> Result<&[u8]>;
502
503    /// Marks the given `amount` of additional bytes from the internal buffer as having been read.
504    /// Subsequent calls to `read` only return bytes that have not been marked as read.
505    ///
506    /// This is a lower-level method and is meant to be used together with [`fill_buf`],
507    /// which can be used to fill the internal buffer via `Read` methods.
508    ///
509    /// It is a logic error if `amount` exceeds the number of unread bytes in the internal buffer, which is returned by [`fill_buf`].
510    ///
511    /// # Examples
512    ///
513    /// Since `consume()` is meant to be used with [`fill_buf`],
514    /// that method's example includes an example of `consume()`.
515    ///
516    /// [`fill_buf`]: BufRead::fill_buf
517    #[stable(feature = "rust1", since = "1.0.0")]
518    fn consume(&mut self, amount: usize);
519
520    /// Checks if there is any data left to be `read`.
521    ///
522    /// This function may fill the buffer to check for data,
523    /// so this function returns `Result<bool>`, not `bool`.
524    ///
525    /// The default implementation calls `fill_buf` and checks that the
526    /// returned slice is empty (which means that there is no data left,
527    /// since EOF is reached).
528    ///
529    /// # Errors
530    ///
531    /// This function will return an I/O error if a `Read` method was called, but returned an error.
532    ///
533    /// Examples
534    ///
535    /// ```
536    /// #![feature(buf_read_has_data_left)]
537    /// use std::io;
538    /// use std::io::prelude::*;
539    ///
540    /// let stdin = io::stdin();
541    /// let mut stdin = stdin.lock();
542    ///
543    /// while stdin.has_data_left()? {
544    ///     let mut line = String::new();
545    ///     stdin.read_line(&mut line)?;
546    ///     // work with line
547    ///     println!("{line:?}");
548    /// }
549    /// # std::io::Result::Ok(())
550    /// ```
551    #[unstable(feature = "buf_read_has_data_left", issue = "86423")]
552    fn has_data_left(&mut self) -> Result<bool> {
553        self.fill_buf().map(|b| !b.is_empty())
554    }
555
556    /// Reads all bytes into `buf` until the delimiter `byte` or EOF is reached.
557    ///
558    /// This function will read bytes from the underlying stream until the
559    /// delimiter or EOF is found. Once found, all bytes up to, and including,
560    /// the delimiter (if found) will be appended to `buf`.
561    ///
562    /// If successful, this function will return the total number of bytes read.
563    ///
564    /// This function is blocking and should be used carefully: it is possible for
565    /// an attacker to continuously send bytes without ever sending the delimiter
566    /// or EOF.
567    ///
568    /// # Errors
569    ///
570    /// This function will ignore all instances of [`ErrorKind::Interrupted`] and
571    /// will otherwise return any errors returned by [`fill_buf`].
572    ///
573    /// If an I/O error is encountered then all bytes read so far will be
574    /// present in `buf` and its length will have been adjusted appropriately.
575    ///
576    /// [`fill_buf`]: BufRead::fill_buf
577    ///
578    /// # Examples
579    ///
580    /// [`std::io::Cursor`][`Cursor`] is a type that implements `BufRead`. In
581    /// this example, we use [`Cursor`] to read all the bytes in a byte slice
582    /// in hyphen delimited segments:
583    ///
584    /// ```
585    /// use std::io::{self, BufRead};
586    ///
587    /// let mut cursor = io::Cursor::new(b"lorem-ipsum");
588    /// let mut buf = vec![];
589    ///
590    /// // cursor is at 'l'
591    /// let num_bytes = cursor.read_until(b'-', &mut buf)
592    ///     .expect("reading from cursor won't fail");
593    /// assert_eq!(num_bytes, 6);
594    /// assert_eq!(buf, b"lorem-");
595    /// buf.clear();
596    ///
597    /// // cursor is at 'i'
598    /// let num_bytes = cursor.read_until(b'-', &mut buf)
599    ///     .expect("reading from cursor won't fail");
600    /// assert_eq!(num_bytes, 5);
601    /// assert_eq!(buf, b"ipsum");
602    /// buf.clear();
603    ///
604    /// // cursor is at EOF
605    /// let num_bytes = cursor.read_until(b'-', &mut buf)
606    ///     .expect("reading from cursor won't fail");
607    /// assert_eq!(num_bytes, 0);
608    /// assert_eq!(buf, b"");
609    /// ```
610    #[stable(feature = "rust1", since = "1.0.0")]
611    fn read_until(&mut self, byte: u8, buf: &mut Vec<u8>) -> Result<usize> {
612        read_until(self, byte, buf)
613    }
614
615    /// Skips all bytes until the delimiter `byte` or EOF is reached.
616    ///
617    /// This function will read (and discard) bytes from the underlying stream until the
618    /// delimiter or EOF is found.
619    ///
620    /// If successful, this function will return the total number of bytes read,
621    /// including the delimiter byte if found.
622    ///
623    /// This is useful for efficiently skipping data such as NUL-terminated strings
624    /// in binary file formats without buffering.
625    ///
626    /// This function is blocking and should be used carefully: it is possible for
627    /// an attacker to continuously send bytes without ever sending the delimiter
628    /// or EOF.
629    ///
630    /// # Errors
631    ///
632    /// This function will ignore all instances of [`ErrorKind::Interrupted`] and
633    /// will otherwise return any errors returned by [`fill_buf`].
634    ///
635    /// If an I/O error is encountered then all bytes read so far will be
636    /// present in `buf` and its length will have been adjusted appropriately.
637    ///
638    /// [`fill_buf`]: BufRead::fill_buf
639    ///
640    /// # Examples
641    ///
642    /// [`std::io::Cursor`][`Cursor`] is a type that implements `BufRead`. In
643    /// this example, we use [`Cursor`] to read some NUL-terminated information
644    /// about Ferris from a binary string, skipping the fun fact:
645    ///
646    /// ```
647    /// use std::io::{self, BufRead};
648    ///
649    /// let mut cursor = io::Cursor::new(b"Ferris\0Likes long walks on the beach\0Crustacean\0!");
650    ///
651    /// // read name
652    /// let mut name = Vec::new();
653    /// let num_bytes = cursor.read_until(b'\0', &mut name)
654    ///     .expect("reading from cursor won't fail");
655    /// assert_eq!(num_bytes, 7);
656    /// assert_eq!(name, b"Ferris\0");
657    ///
658    /// // skip fun fact
659    /// let num_bytes = cursor.skip_until(b'\0')
660    ///     .expect("reading from cursor won't fail");
661    /// assert_eq!(num_bytes, 30);
662    ///
663    /// // read animal type
664    /// let mut animal = Vec::new();
665    /// let num_bytes = cursor.read_until(b'\0', &mut animal)
666    ///     .expect("reading from cursor won't fail");
667    /// assert_eq!(num_bytes, 11);
668    /// assert_eq!(animal, b"Crustacean\0");
669    ///
670    /// // reach EOF
671    /// let num_bytes = cursor.skip_until(b'\0')
672    ///     .expect("reading from cursor won't fail");
673    /// assert_eq!(num_bytes, 1);
674    /// ```
675    #[stable(feature = "bufread_skip_until", since = "1.83.0")]
676    fn skip_until(&mut self, byte: u8) -> Result<usize> {
677        skip_until(self, byte)
678    }

680    /// Reads all bytes until a newline (the `0xA` byte) is reached, and append
681    /// them to the provided `String` buffer.
682    ///
683    /// Previous content of the buffer will be preserved. To avoid appending to
684    /// the buffer, you need to [`clear`] it first.
685    ///
686    /// This function will read bytes from the underlying stream until the
687    /// newline delimiter (the `0xA` byte) or EOF is found. Once found, all bytes
688    /// up to, and including, the delimiter (if found) will be appended to
689    /// `buf`.
690    ///
691    /// If successful, this function will return the total number of bytes read.
692    ///
693    /// If this function returns [`Ok(0)`], the stream has reached EOF.
694    ///
695    /// This function is blocking and should be used carefully: it is possible for
696    /// an attacker to continuously send bytes without ever sending a newline
697    /// or EOF. You can use [`take`] to limit the maximum number of bytes read.
698    ///
699    /// [`Ok(0)`]: Ok
700    /// [`clear`]: String::clear
701    /// [`take`]: crate::io::Read::take
702    ///
703    /// # Errors
704    ///
705    /// This function has the same error semantics as [`read_until`] and will
706    /// also return an error if the read bytes are not valid UTF-8. If an I/O
707    /// error is encountered then `buf` may contain some bytes already read in
708    /// the event that all data read so far was valid UTF-8.
709    ///
710    /// [`read_until`]: BufRead::read_until
711    ///
712    /// # Examples
713    ///
714    /// [`std::io::Cursor`][`Cursor`] is a type that implements `BufRead`. In
715    /// this example, we use [`Cursor`] to read all the lines in a byte slice:
716    ///
717    /// ```
718    /// use std::io::{self, BufRead};
719    ///
720    /// let mut cursor = io::Cursor::new(b"foo\nbar");
721    /// let mut buf = String::new();
722    ///
723    /// // cursor is at 'f'
724    /// let num_bytes = cursor.read_line(&mut buf)
725    ///     .expect("reading from cursor won't fail");
726    /// assert_eq!(num_bytes, 4);
727    /// assert_eq!(buf, "foo\n");
728    /// buf.clear();
729    ///
730    /// // cursor is at 'b'
731    /// let num_bytes = cursor.read_line(&mut buf)
732    ///     .expect("reading from cursor won't fail");
733    /// assert_eq!(num_bytes, 3);
734    /// assert_eq!(buf, "bar");
735    /// buf.clear();
736    ///
737    /// // cursor is at EOF
738    /// let num_bytes = cursor.read_line(&mut buf)
739    ///     .expect("reading from cursor won't fail");
740    /// assert_eq!(num_bytes, 0);
741    /// assert_eq!(buf, "");
742    /// ```
743    #[stable(feature = "rust1", since = "1.0.0")]
744    fn read_line(&mut self, buf: &mut String) -> Result<usize> {
745        // Note that we are not calling the `.read_until` method here, but
746        // rather our hardcoded implementation. For more details as to why, see
747        // the comments in `default_read_to_string`.
748        unsafe { append_to_string(buf, |b| read_until(self, b'\n', b)) }
749    }

751    /// Returns an iterator over the contents of this reader split on the byte
752    /// `byte`.
753    ///
754    /// The iterator returned from this function will return instances of
755    /// <code>[io::Result]<[Vec]\<u8>></code>. Each vector returned will *not* have
756    /// the delimiter byte at the end.
757    ///
758    /// This function will yield errors whenever [`read_until`] would have
759    /// also yielded an error.
760    ///
761    /// [io::Result]: self::Result "io::Result"
762    /// [`read_until`]: BufRead::read_until
763    ///
764    /// # Examples
765    ///
766    /// [`std::io::Cursor`][`Cursor`] is a type that implements `BufRead`. In
767    /// this example, we use [`Cursor`] to iterate over all hyphen delimited
768    /// segments in a byte slice
769    ///
770    /// ```
771    /// use std::io::{self, BufRead};
772    ///
773    /// let cursor = io::Cursor::new(b"lorem-ipsum-dolor");
774    ///
775    /// let mut split_iter = cursor.split(b'-').map(|l| l.unwrap());
776    /// assert_eq!(split_iter.next(), Some(b"lorem".to_vec()));
777    /// assert_eq!(split_iter.next(), Some(b"ipsum".to_vec()));
778    /// assert_eq!(split_iter.next(), Some(b"dolor".to_vec()));
779    /// assert_eq!(split_iter.next(), None);
780    /// ```
781    #[stable(feature = "rust1", since = "1.0.0")]
782    fn split(self, byte: u8) -> Split<Self>
783    where
784        Self: Sized,
785    {
786        Split { buf: self, delim: byte }
787    }

789    /// Returns an iterator over the lines of this reader.
790    ///
791    /// The iterator returned from this function will yield instances of
792    /// <code>[io::Result]<[String]></code>. Each string returned will *not* have a newline
793    /// byte (the `0xA` byte) or `CRLF` (`0xD`, `0xA` bytes) at the end.
794    ///
795    /// [io::Result]: self::Result "io::Result"
796    ///
797    /// # Examples
798    ///
799    /// [`std::io::Cursor`][`Cursor`] is a type that implements `BufRead`. In
800    /// this example, we use [`Cursor`] to iterate over all the lines in a byte
801    /// slice.

811    /// assert_eq!(lines_iter.next(), Some(String::from("dolor")));
812    /// assert_eq!(lines_iter.next(), None);

815    /// # Errors

817    /// Each line of the iterator has the same error semantics as [`BufRead::read_line`].

819    fn lines(self) -> Lines<Self>
820    where
821        Self: Sized,
822    {
823        Lines { buf: self }
824    }

827#[stable(feature = "chain_bufread", since = "1.9.0")]
828impl<T: BufRead, U: BufRead> BufRead for Chain<T, U> {
829    fn fill_buf(&mut self) -> Result<&[u8]> {
830        if !self.done_first {
831            match self.first.fill_buf()? {
832                buf if buf.is_empty() => self.done_first = true,
833                buf => return Ok(buf),
834            }
835        }
836        self.second.fill_buf()
837    }
838
839    fn consume(&mut self, amt: usize) {
840        if !self.done_first { self.first.consume(amt) } else { self.second.consume(amt) }
841    }
842
843    fn read_until(&mut self, byte: u8, buf: &mut Vec<u8>) -> Result<usize> {
844        let mut read = 0;
845        if !self.done_first {
846            let n = self.first.read_until(byte, buf)?;
847            read += n;
848
849            match buf.last() {
850                Some(b) if *b == byte && n != 0 => return Ok(read),
851                _ => self.done_first = true,
852            }
853        }
854        read += self.second.read_until(byte, buf)?;
855        Ok(read)
856    }
857
858    // We don't override `read_line` here because an UTF-8 sequence could be
859    // split between the two parts of the chain
860}
861
862#[stable(feature = "rust1", since = "1.0.0")]
863impl<T: BufRead> BufRead for Take<T> {
864    fn fill_buf(&mut self) -> Result<&[u8]> {
865        // Don't call into inner reader at all at EOF because it may still block
866        if self.limit == 0 {
867            return Ok(&[]);
868        }
869
870        let buf = self.inner.fill_buf()?;
871        let cap = cmp::min(buf.len() as u64, self.limit) as usize;
872        Ok(&buf[..cap])
873    }
874
875    fn consume(&mut self, amt: usize) {
876        // Don't let callers reset the limit by passing an overlarge value
877        let amt = cmp::min(amt as u64, self.limit) as usize;
878        self.limit -= amt as u64;
879        self.inner.consume(amt);
880    }
881}

883/// An iterator over the contents of an instance of `BufRead` split on a
884/// particular byte.
885///
886/// This struct is generally created by calling [`split`] on a `BufRead`.
887/// Please see the documentation of [`split`] for more details.
888///
889/// [`split`]: BufRead::split
890#[stable(feature = "rust1", since = "1.0.0")]
891#[derive(Debug)]
892#[cfg_attr(not(test), rustc_diagnostic_item = "IoSplit")]
893pub struct Split<B> {
894    buf: B,
895    delim: u8,
896}
897
898#[stable(feature = "rust1", since = "1.0.0")]
899impl<B: BufRead> Iterator for Split<B> {
900    type Item = Result<Vec<u8>>;
901
902    fn next(&mut self) -> Option<Result<Vec<u8>>> {
903        let mut buf = Vec::new();
904        match self.buf.read_until(self.delim, &mut buf) {
905            Ok(0) => None,
906            Ok(_n) => {
907                if buf[buf.len() - 1] == self.delim {
908                    buf.pop();
909                }
910                Some(Ok(buf))
911            }
912            Err(e) => Some(Err(e)),
913        }
914    }
915}

918///
919/// This struct is generally created by calling [`lines`] on a `BufRead`.
920/// Please see the documentation of [`lines`] for more details.
921///
922/// [`lines`]: BufRead::lines

928}

930#[stable(feature = "rust1", since = "1.0.0")]
931impl<B: BufRead> Iterator for Lines<B> {
932    type Item = Result<String>;
933
934    fn next(&mut self) -> Option<Result<String>> {
935        let mut buf = String::new();
936        match self.buf.read_line(&mut buf) {
937            Ok(0) => None,
938            Ok(_n) => {
939                if buf.ends_with('\n') {
940                    buf.pop();
941                    if buf.ends_with('\r') {
942                        buf.pop();
943                    }
944                }
945                Some(Ok(buf))
946            }
947            Err(e) => Some(Err(e)),
948        }
949    }
